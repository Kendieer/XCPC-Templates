\section{并查集}

\subsection{普通版本}

单次操作均摊时间复杂度为 $\alpha(n)$，其中 $\alpha$ 为反阿克曼函数，在 $10^6$ 量级下值大约为 $6$。

\texttt{Disjoint} 用整数 $0\dots n$ 各建独立集合（各元素为代表）。调用 \texttt{d(u,v)}（重载 \texttt{operator()}）合并 $u$、$v$ 所在集合，合并后新集合的代表取第一个参数一侧的原代表；\texttt{d[x]}（重载 \texttt{operator[]}）返回 $x$ 所在集合的代表值，\texttt{size(x)} 返回其大小，\texttt{init(n)} 可重建。适合需要“代表值”而不仅是判连通的合并场景（如按起点方向合并相邻连通块后询问块首）。

\begin{lstlisting}
struct Disjoint{
    std::vector<int> fa,siz,top;
    int cnt=0;
    Disjoint(int n = 0){
        init(n);
    }
    void init(int n = 0){
        fa.resize(n+1);
        top.resize(n+1);
        siz.assign(n+1,1);
        std::iota(fa.begin(),fa.end(),0);
        std::iota(top.begin(),top.end(),0);
        cnt = n;
    }
    void operator()(int head,int ele){ // merge
        head = root(head),ele = root(ele);
        if (head == ele) return;
        top[ele] = top[head];
        if (siz[head] < siz[ele]) std::swap(head,ele);
        siz[head]+=siz[ele];
        fa[ele]=head;
    }
    int operator[](int index){return top[root(index)];}
    int size(int x){return siz[root(x)];}
private:
    int root(int x){
        return fa[x] = fa[x] == x ? x : root(fa[x]);
    }
};
\end{lstlisting}

\subsection{可撤销版本}

\noindent 支持按顺序撤销合并的版本（离线分治、动态图连通性等场景）。\texttt{d(x,y)} 合并并返回是否真的发生了合并；\texttt{d[x]} 返回 $x$ 所在集合的根；\texttt{undo(cnt)} 撤销最近 \texttt{cnt} 次确实生效的合并。因需回滚而不做路径压缩、只按大小合并，单次合并 $\mathcal{O}(\log n)$，\texttt{undo} 代价 $\mathcal{O}(\textit{cnt})$。

\begin{lstlisting}
struct DisjointWithUndo{
    std::vector<int> siz,fa;
    std::vector<std::array<int,2>> tm;// fa,son
    DisjointWithUndo(int n = 0){init(n);}
    void init(int n){
        siz = std::vector<int>(n+1,1),fa = std::vector<int>(n+1),tm.clear();
        std::iota(fa.begin(),fa.end(),0);
    }
    bool operator()(int x,int y){ // 合并
        x = root(x),y = root(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) std::swap(x,y);
        tm.push_back({x,y});
        siz[x] += siz[y],fa[y] = x;
        return true;
    }
    int operator[](int index){return root(index);}
    void undo(int cnt){ // 撤销
        while(cnt--){
            if (tm.empty()) break;
            auto [x,y] = tm.back();
            tm.pop_back();
            siz[x] -= siz[y];
            fa[y] = y;
        }
    }
private:
    int root(int x){
        while(x!=fa[x]) x=fa[x];
        return x;
    }
};
\end{lstlisting}
