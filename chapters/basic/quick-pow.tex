\section{快速幂}

\verb|qpow(a,b,mod)| 计算 $a^b \bmod mod$，复杂度 $\mathcal{O}(\log b)$。

\subsection{整数快速幂}

\begin{lstlisting}
long long qpow(long long a, long long b, long long mod) {
    long long res = 1;
    for (; b; b >>= 1) {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
    }
    return res;
}
\end{lstlisting}

\subsection{大模数防溢出}

当 $mod$ 接近 $10^{18}$ 量级时 \texttt{res * a} 会溢出 \texttt{long long}，改用 \verb|mulmod(a,b,mod)|（借助 \texttt{\_\_int128} 的不溢出模乘）；\verb|qpow| 用法同上，仅内部换成 \verb|mulmod|。

\begin{lstlisting}
long long mulmod(long long a, long long b, long long mod) {
    return (__int128)a * b % mod;
}
long long qpow(long long a, long long b, long long mod) {
    long long res = 1;
    for (; b; b >>= 1) {
        if (b & 1) res = mulmod(res, a, mod);
        a = mulmod(a, a, mod);
    }
    return res;
}
\end{lstlisting}
